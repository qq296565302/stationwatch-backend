### 2.4 TypeORM 实体示例（duty_records.entity.ts）

```typescript
import { Entity, PrimaryGeneratedColumn, Column, CreateDateColumn, UpdateDateColumn, OneToMany, ManyToOne, Unique, JoinColumn, Index } from 'typeorm';
import { Station } from '../stations/station.entity';
import { User } from '../users/user.entity';
import { DutyItem } from '../duty-items/duty-item.entity';

@Entity('duty_records')
@Unique(['station', 'recordDate'])
@Index(['creator'])
@Index(['status'])
export class DutyRecord {
  @PrimaryGeneratedColumn({ type: 'bigint', unsigned: true })
  id: string;

  @Column({ name: 'record_date', type: 'date' })
  recordDate: string;

  @ManyToOne(() => Station, { eager: true })
  @JoinColumn({ name: 'station_id' })
  station: Station;

  @Column({ length: 20 })
  weather: string;

  @Column({ name: 'weather_label', length: 20 })
  weatherLabel: string;

  @ManyToOne(() => User)
  @JoinColumn({ name: 'creator_id' })
  creator: User;

  @Column({ type: 'enum', enum: ['active', 'locked', 'archived'], default: 'active' })
  status: 'active' | 'locked' | 'archived';

  @Column({ name: 'item_count', default: 0 })
  itemCount: number;

  @Column({ name: 'completed_count', default: 0 })
  completedCount: number;

  @Column({ name: 'has_pending', default: false })
  hasPending: boolean;

  @Column({ name: 'other_matters', type: 'text', nullable: true })
  otherMatters: string;

  @Column({ name: 'pending_issues', type: 'text', nullable: true })
  pendingIssues: string;

  @Column({ name: 'locked_at', type: 'datetime', nullable: true })
  lockedAt: Date;

  @ManyToOne(() => User, { nullable: true })
  @JoinColumn({ name: 'locked_by' })
  lockedBy: User;

  @OneToMany(() => DutyItem, item => item.record, { cascade: true })
  items: DutyItem[];

  @CreateDateColumn({ name: 'created_at' })
  createdAt: Date;

  @UpdateDateColumn({ name: 'updated_at' })
  updatedAt: Date;
}
```

### 3.3 核心接口详细定义

#### 3.3.1 POST /records（智能 upsert）

**入参**：
```typescript
interface DutyRecordUpsertDto {
  recordDate: string;           // '2026-07-30'
  stationId: number;
  weather: string;              // 'sunny' | 'cloudy' | 'rainy' | 'stormy' | 'snowy' | 'foggy'
  weatherLabel: string;         // '晴天'
  dutyItems: DutyItemUpsertDto[];
  otherMatters?: string;        // 追加，不覆盖
  pendingIssues?: string;       // 追加
}

interface DutyItemUpsertDto {
  id?: number;                  // 有 id 则更新，无 id 则追加
  businessType: string;
  content: string;
  acceptTime?: string;          // 'HH:MM'
  endTime?: string;
  customerName?: string;
  customerPhone?: string;
  customerAddress?: string;
  handler?: string;
  result?: string;
  isCompleted?: boolean;
}
```

**业务逻辑**：
1. 开启事务 + 行锁 `SELECT ... FOR UPDATE`
2. 查找 `WHERE station_id = ? AND record_date = ?`
3. **存在**：合并工单
   - 有 `id` 且在 DB 存在 → UPDATE
   - 无 `id` 或不在 DB → INSERT（用前端临时 id 也行，后端给最终 id）
   - `other_matters` 用 `\n` 追加
   - `pending_issues` 同样追加
   - 重算 `item_count / completed_count / has_pending`
4. **不存在**：INSERT
5. 返回合并后的完整记录（包含 `items` 数组）

**返回**：
```json
{
  "code": 0,
  "data": { /* 完整 duty_record 实体 */ }
}
```

#### 3.3.2 POST /records/:recordId/items

**入参**：
```typescript
interface CreateDutyItemDto {
  businessType: string;
  content: string;
  customerName?: string;
  customerPhone?: string;
  customerAddress?: string;
  handler?: string;
  result?: string;
}
```

**业务逻辑**：
- 验证记录存在 + 未锁定
- 验证 `currentItemCount < station.max_duty_items_per_record`（默认 11）
- `acceptTime` 自动设 `HH:MM`（如果没传）
- INSERT
- 更新记录的 `item_count`

#### 3.3.3 POST /records/:recordId/items/:itemId/complete

**入参**：无

**业务逻辑**：
- 验证记录未锁定
- `endTime = now()`（**强制覆盖**，即使前端传了）
- `isCompleted = true`
- 更新记录的 `completed_count`

**入参**：

```typescript
@Injectable()
export class DutyRecordsService {
  constructor(
    @InjectRepository(DutyRecord)
    private recordsRepo: Repository<DutyRecord>,
    @InjectRepository(DutyItem)
    private itemsRepo: Repository<DutyItem>,
  ) {}

  async upsert(dto: DutyRecordUpsertDto, user: User): Promise<DutyRecord> {
    return this.recordsRepo.manager.transaction(async manager => {
      // 1. 查找现有记录（带锁）
      let record = await manager
        .createQueryBuilder(DutyRecord, 'r')
        .setLock('pessimistic_write')
        .where('r.station_id = :stationId AND r.record_date = :date', {
          stationId: dto.stationId,
          date: dto.recordDate,
        })
        .getOne();

      if (record) {
        // 2. 验证权限
        if (record.status === 'locked' && user.role !== 'admin') {
          throw new BusinessException(20003, '记录已锁定，无法编辑');
        }
      }

      if (!record) {
        // 3. 新建
        record = manager.create(DutyRecord, {
          recordDate: dto.recordDate,
          station: { id: dto.stationId } as Station,
          weather: dto.weather,
          weatherLabel: dto.weatherLabel,
          creator: { id: user.id } as User,
          status: 'active',
          otherMatters: dto.otherMatters || '',
          pendingIssues: dto.pendingIssues || '',
        });
        record = await manager.save(record);
      } else {
        // 4. 更新基本信息
        record.weather = dto.weather;
        record.weatherLabel = dto.weatherLabel;
        // 追加文本字段
        if (dto.otherMatters?.trim()) {
          record.otherMatters = record.otherMatters
            ? `${record.otherMatters}\n${dto.otherMatters}`
            : dto.otherMatters;
        }
        if (dto.pendingIssues?.trim()) {
          record.pendingIssues = record.pendingIssues
            ? `${record.pendingIssues}\n${dto.pendingIssues}`
            : dto.pendingIssues;
        }
        await manager.save(record);
      }

      // 5. 合并工单
      const existingItems = await manager.find(DutyItem, { where: { record: { id: record.id } } });
      const byId = new Map(existingItems.map(i => [i.id, i]));

      for (const itemDto of dto.dutyItems) {
        if (itemDto.id && byId.has(itemDto.id)) {
          // 更新
          const existing = byId.get(itemDto.id)!;
          Object.assign(existing, {
            businessType: itemDto.businessType,
            content: itemDto.content,
            acceptTime: itemDto.acceptTime ?? existing.acceptTime,
            endTime: itemDto.endTime ?? existing.endTime,
            customerName: itemDto.customerName,
            customerPhone: itemDto.customerPhone,
            customerAddress: itemDto.customerAddress,
            handler: itemDto.handler,
            result: itemDto.result,
            isCompleted: itemDto.isCompleted ?? existing.isCompleted,
          });
          await manager.save(existing);
        } else {
          // 追加
          const newItem = manager.create(DutyItem, {
            record: { id: record.id } as DutyRecord,
            businessType: itemDto.businessType,
            content: itemDto.content,
            acceptTime: itemDto.acceptTime || this.formatNow(),
            endTime: itemDto.endTime,
            customerName: itemDto.customerName,
            customerPhone: itemDto.customerPhone,
            customerAddress: itemDto.customerAddress,
            handler: itemDto.handler,
            result: itemDto.result,
            isCompleted: itemDto.isCompleted ?? false,
            sortOrder: existingItems.length,
          });
          await manager.save(newItem);
        }
      }

      // 6. 重算冗余字段
      const allItems = await manager.find(DutyItem, { where: { record: { id: record.id } } });
      record.itemCount = allItems.length;
      record.completedCount = allItems.filter(i => i.isCompleted).length;
      record.hasPending = !!record.pendingIssues?.trim();
      await manager.save(record);

      return this.findOne(record.id);
    });
  }

  private formatNow(): string {
    const now = new Date();
    return `${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}`;
  }
}
```

```typescript
  async generateMonthly(opts: {
    year: number;
    month: number;
    stationId: number;
    includeTemplate: boolean;
    mergeSheets: boolean;
  }): Promise<ExportHistory> {
    // 1. 查数据

          `${opts.year}-${String(opts.month).padStart(2, '0')}-01`,

        ),
      },
      relations: ['items'],
      order: { recordDate: 'ASC' },
    });

    // 2. 创建工作簿
    const wb = new Workbook();

    if (opts.includeTemplate) {
      // 从模板复制样式
      const template = await wb.xlsx.readFile('templates/电子值班表模板.xlsx');
      const tplSheet = template.worksheets[0];
      wb.addWorksheet('模板说明');
      // 复制模板内容
    }

      this.fillSheetHeader(ws, opts.year, opts.month);
      records.forEach((r, idx) => this.fillRecord(ws, r, idx + 2));
    } else {
      // 每天一个 sheet
      records.forEach(r => {
        const ws = wb.addWorksheet(this.formatDate(r.recordDate));
        this.fillSheetHeader(ws, r.recordDate, r.weather);
        this.fillRecord(ws, r, 2);
      });
    }

    // 3. 保存
    const fileName = `值班记录_${opts.year}-${String(opts.month).padStart(2, '0')}.xlsx`;
    const filePath = path.join(process.env.EXPORT_DIR, fileName);
    await wb.xlsx.writeFile(filePath);

    // 4. 写历史
    const stat = await fs.stat(filePath);
    return this.historyRepo.save({
      fileName,
      fileSize: stat.size,
      filePath,
      year: opts.year,
      month: opts.month,
      station: { id: opts.stationId } as Station,
      operator: { id: 1 } as User,
      status: 'done',
    });
  }

  private fillRecord(ws: any, record: DutyRecord, startRow: number) {
    let row = startRow;
    // 值班信息

    ws.getCell(`B${row}`).value = record.weatherLabel;
    ws.getCell(`C${row}`).value = record.creator.realName;
    row++;
    // 工单

      ws.getCell(`B${row}`).value = item.businessType;
      ws.getCell(`C${row}`).value = item.content;
      ws.getCell(`D${row}`).value = item.customerName;
      ws.getCell(`E${row}`).value = item.customerPhone;
      ws.getCell(`F${row}`).value = item.handler;
      ws.getCell(`G${row}`).value = item.result;
      ws.getCell(`H${row}`).value = item.isCompleted ? '✓' : '';
      row++;
    });
  }
```

```typescript
// auth.service.ts
@Injectable()
export class AuthService {
  constructor(
    private jwtService: JwtService,
    private redisService: RedisService,
  ) {}

  async login(username: string, password: string) {
    const user = await this.usersService.findByUsername(username);
    if (!user || !await bcrypt.compare(password, user.passwordHash)) {
      throw new UnauthorizedException('账号或密码错误');
    }
    if (!user.isActive) throw new UnauthorizedException('账号已停用');

    const accessToken = this.jwtService.sign(
      { sub: user.id, role: user.role, stationId: user.stationId },
      { expiresIn: '2h' },
    );
    const refreshToken = this.jwtService.sign(
      { sub: user.id, type: 'refresh' },
      { expiresIn: '7d' },
    );

    // refreshToken 存 Redis（7 天）
    await this.redisService.set(
      `refresh:${user.id}`,
      refreshToken,
      7 * 24 * 60 * 60,
    );

    // 记录登录时间
    user.lastLoginAt = new Date();
    user.lastLoginIp = ...;
    await this.usersRepo.save(user);

    return { accessToken, refreshToken, user: this.sanitizeUser(user) };
  }

  async refresh(refreshToken: string) {
    let payload: any;
    try {
      payload = this.jwtService.verify(refreshToken);
    } catch (e) {
      throw new UnauthorizedException('refresh token 无效');
    }
    if (payload.type !== 'refresh') {
      throw new UnauthorizedException('token 类型错误');
    }

    // 验证 Redis 中的 token 一致
    const stored = await this.redisService.get(`refresh:${payload.sub}`);
    if (stored !== refreshToken) {
      throw new UnauthorizedException('refresh token 已失效');
    }

    const user = await this.usersService.findById(payload.sub);
    const accessToken = this.jwtService.sign(
      { sub: user.id, role: user.role, stationId: user.stationId },
      { expiresIn: '2h' },
    );
    return { accessToken };
  }

  async logout(userId: number) {
    await this.redisService.del(`refresh:${userId}`);
    return { ok: true };
  }
}
```

```typescript
async search(q: string, stationId: number, limit = 20) {
  const pattern = `%${q}%`;

  const records = await this.recordsRepo
    .createQueryBuilder('r')
    .leftJoinAndSelect('r.station', 'station')
    .where('r.station_id = :stationId', { stationId })
    .andWhere(new Brackets(qb => {
      qb.where('r.other_matters LIKE :p', { p: pattern })
        .orWhere('r.pending_issues LIKE :p', { p: pattern })
        .orWhere('r.weather_label LIKE :p', { p: pattern });
    }))
    .getMany();

  const items = await this.itemsRepo
    .createQueryBuilder('i')
    .innerJoinAndSelect('i.record', 'r')
    .where('r.station_id = :stationId', { stationId })
    .andWhere(new Brackets(qb => {
      qb.where('i.content LIKE :p', { p: pattern })
        .orWhere('i.business_type LIKE :p', { p: pattern })
        .orWhere('i.customer_name LIKE :p', { p: pattern })
        .orWhere('i.customer_phone LIKE :p', { p: pattern })
        .orWhere('i.customer_address LIKE :p', { p: pattern })
        .orWhere('i.handler LIKE :p', { p: pattern })
        .orWhere('i.result LIKE :p', { p: pattern });
    }))
    .getMany();

  const results = [];
  records.forEach(r => results.push({
    type: 'record',
    recordId: r.id,
    recordDate: r.recordDate,
    title: `${r.recordDate} · ${r.station.name}`,
    snippet: r.otherMatters || r.pendingIssues || '',
  }));
  items.forEach(i => results.push({
    type: 'item',
    recordId: i.record.id,
    recordDate: i.record.recordDate,
    itemId: i.id,
    businessType: i.businessType,
    title: i.content,
```

### 7.2 main.ts 入口

```typescript
import { NestFactory } from '@nestjs/core';
import { ValidationPipe } from '@nestjs/common';
import { SwaggerModule, DocumentBuilder } from '@nestjs/swagger';
import { AppModule } from './app.module';
import { AllExceptionsFilter } from './common/filters/all-exceptions.filter';
import { TransformInterceptor } from './common/interceptors/transform.interceptor';

async function bootstrap() {
  const app = await NestFactory.create(AppModule, { bufferLogs: true });

  // CORS
  app.enableCors({
    origin: process.env.CORS_ORIGIN?.split(',') || '*',
    credentials: true,
  });

  // 全局前缀
  app.setGlobalPrefix('api/v1');

  // 全局校验
  app.useGlobalPipes(new ValidationPipe({
    whitelist: true,
    forbidNonWhitelisted: true,
    transform: true,
  }));

  // 全局过滤器
  app.useGlobalFilters(new AllExceptionsFilter());

  // 全局拦截器
  app.useGlobalInterceptors(new TransformInterceptor());

  // Swagger
  if (process.env.SWAGGER_ENABLED === 'true') {
    const config = new DocumentBuilder()
      .setTitle('供电所值守云平台 API')
      .setDescription('后端接口文档')
      .setVersion('1.0')
      .addBearerAuth()
      .build();
    const document = SwaggerModule.createDocument(app, config);
    SwaggerModule.setup(process.env.SWAGGER_PATH || 'api/docs', app, document);
  }

  await app.listen(process.env.PORT || 3000);
  console.log(`🚀 Application is running on: http://localhost:${process.env.PORT || 3000}/api/v1`);
}
bootstrap();
```

## 九、开发流程

### 9.1 推荐开发顺序

| 阶段 | 任务 | 说明 |

| 4 | 种子数据 | 站点、用户、字典 |
| 5 | 认证模块 | login / refresh / logout / me |
| 6 | 用户/站点 CRUD | 配合前端开发 |
| 7 | 值班记录 upsert | 核心：合并工单 |
| 8 | 值班工单 CRUD | 自动时间 |

| 18 | 部署 | Docker + Nginx |

### 9.2 开发规范

- **TypeScript strict** 全开
- **DTO 必填校验**：所有入参都用 class-validator
- **Service 抛业务异常**：`throw new BusinessException(code, message)`，由全局过滤器转响应
- **Service 不直接返回 entity**：用 mapper 转为 DTO（避免循环引用和敏感字段泄露）
- **repository 写在 service 内**：不开 controller 访问权限
- **事务**：多表操作必须用 `manager.transaction`
- **乐观锁/悲观锁**：upsert 场景必须用悲观锁防并发
- **日志**：service 内 `this.logger.log/debug/warn`，不用 console
- **环境变量**：所有配置走 .env，不用 hardcode

### 11.2 前端 store 改造示例

```typescript
// store/index.ts (axios 版)
import axios from 'axios';
import { defineStore } from 'pinia';

const api = axios.create({
  baseURL: import.meta.env.VITE_API_BASE || '/api/v1',
  timeout: 15000,
});

api.interceptors.request.use((config) => {
  const auth = JSON.parse(localStorage.getItem('dutyguard_auth') || '{}');
  if (auth.accessToken) {
    config.headers.Authorization = `Bearer ${auth.accessToken}`;
  }
  return config;
});

api.interceptors.response.use(
  (resp) => resp.data.data,
  async (error) => {
    if (error.response?.status === 401) {
      // token 过期，尝试 refresh
      const auth = JSON.parse(localStorage.getItem('dutyguard_auth') || '{}');
      if (auth.refreshToken) {
        try {
          const { data } = await axios.post('/api/v1/auth/refresh', {
            refreshToken: auth.refreshToken,
          });
          // 更新 token
          localStorage.setItem('dutyguard_auth', JSON.stringify({
            ...auth,
            accessToken: data.data.accessToken,
          }));
          // 重试原请求
          error.config.headers.Authorization = `Bearer ${data.data.accessToken}`;
          return api.request(error.config);
        } catch (e) {
          // refresh 失败，跳登录
          localStorage.removeItem('dutyguard_auth');
          window.location.href = '/login';
        }
      }
    }
    throw error;
  },
);

export const useAppStore = defineStore('app', {
  state: () => ({
    user: null,
    isLoggedIn: false,
    records: [],
    dictionaries: null,
  }),
  actions: {
    async login(username, password) {
      const data = await api.post('/auth/login', { username, password });
      this.user = data.user;
      this.isLoggedIn = true;
      localStorage.setItem('dutyguard_auth', JSON.stringify({
        user: data.user,
        accessToken: data.accessToken,
        refreshToken: data.refreshToken,
        isLoggedIn: true,
      }));
    },
    async createRecord(recordData) {
      const saved = await api.post('/records', recordData);
      const idx = this.records.findIndex(r => r.id === saved.id);
      if (idx >= 0) this.records[idx] = saved;
      else this.records.unshift(saved);
      return saved;
    },
    // ... 其他 action 同步改造
  },
});
```
